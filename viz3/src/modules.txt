# Module list of the app, in bundle order (build.py concatenates them into one shared scope: later modules see everything above).
# The order matters: modules run at load in this sequence (meshes before render/zmerge.js get merged; point lights are created in pavilion/interior.js
# and garden/workshop_lamp.js). A new module goes where its dependencies are above it – usually before boot/boot.js.

core/head.js                # three.js imports, renderer, scene, camera, query flags, small helpers
core/assets.js              # loading manager, textures, PBR materials, patchSV (sky visibility, baked GI, figure shadows, lamp-spill mask)
core/geo.js                 # box/cylinder builders with collision + walk registry (COL, WALK), building registry (bldBegin/bldEnd, BUILD)
core/mats.js                # shared materials MAT
core/detail.js              # V16b1 detail textures of the ground shaders as one texture array (DETAIL); sampler count check and fallback (GLCHECK)
pavilion/house_base.js      # the planned pavilion: dimensions, levels, wall/window/railing/ladder helpers
pavilion/house_common.js    # parts both variants share (slab, EG walls S/N/W, joists, chimney, EG windows, floor, stove)
pavilion/house_tower.js     # variant A: stair tower
pavilion/house_ladder.js    # variant B: ladder stairs, trapdoor, roof hatch
pavilion/interior.js        # furniture, lamps (LIGHTS) – shared and variant A
pavilion/interior_ladder.js # variant B furnishing
garden/stream_geo.js        # the stream: shared geometry (STREAM)
garden/env.js               # sky dome, sun light + shadow box, ground, far terrain, neighbour sheds

game/game.js                # walking: collision, doors, player state, camera, input, frame loop
